# The Motorola MC6809's instruction-set description: what the assembler, disassembler and tools know about it.
#
# Native syntax writes indirection in brackets ([...]) so parentheses always group expressions: LDA [,X],
# LDA [5,Y], LDA [,U++], LDA [Label,PCR] and LDA [$1234]. Direct-page versus extended addressing is chosen by
# value ($00-$FF uses direct when unforced), or forced with < and > as Motorola does.
#
# The undocumented encodings follow David Banks's Undocumented 6809 Behaviours
# (https://github.com/hoglet67/6809Decoder/wiki/Undocumented-6809-Behaviours) and David Flamand's Comprehensive
# Document of 6809 Undocumented and Undefined Behavior, both from real chips, with Banks's names. A prefix before
# an opcode it leaves undefined is ignored, so those pairs are not listed.

[isa]
endian big
address-bits 16
angle-widths
registers A:8 B:8 D:16 X:16 Y:16 U:16 S:16 DP:8 CC:8 PC:16
flags CC E=7 F=6 H=5 I=4 N=3 Z=2 V=1 C=0
flag E entire "Set when the complete register state was stacked for an interrupt, so RTI pulls all registers; cleared by FIRQ so RTI pulls only CC and PC."
flag F fast-interrupt "Set to mask FIRQ interrupts, by reset, FIRQ, NMI and SWI."
flag H half-carry "Set when an 8-bit add carries out of bit 3; DAA uses it to correct packed BCD results."
flag I interrupt "Set to mask IRQ interrupts, by reset, IRQ, FIRQ, NMI and SWI."
flag N negative "Set when a result's most significant bit is 1, a negative number in two's complement."
flag Z zero "Set when a result is zero, or when a compare finds the values equal."
flag V overflow "Set when signed arithmetic overflows two's-complement range."
flag C carry "Set when an add carries out of the top bit, a subtract or compare borrows, or MUL sets bit 7 of B. Shifts and rotates move bits through it."
alias LSL=ASL LSLA=ASLA LSLB=ASLB BHS=BCC BLO=BCS LBHS=LBCC LBLO=LBCS
vectors FFFE FFFC FFFA FFF8 FFF6 FFF4 FFF2

# Control flow, for disassembly and analysis.
flow branch BHI BLS BCC BCS BNE BEQ BVC BVS BPL BMI BGE BLT BGT BLE LBHI LBLS LBCC LBCS LBNE LBEQ LBVC LBVS LBPL LBMI LBGE LBLT LBGT LBLE
flow jump BRA LBRA JMP
flow indirect-jump JMP:Indexed
flow call BSR LBSR JSR
flow return RTS RTI
flow wait SYNC CWAI
flow stop XHCF

# How an instruction accesses a memory operand; the rest read it.
access write STA STB STD STX STY STU STS CLR
access read-write NEG COM LSR ROR ASR ASL ROL DEC INC XNC XDEC
