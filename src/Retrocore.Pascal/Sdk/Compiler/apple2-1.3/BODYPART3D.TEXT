(** BODYPART3D.TEXT: BODYPART's code generator, continued.

    Included by COMPILER.TEXT after BODYPART3C.TEXT and before BODYPART4A.TEXT.

    Procedures here, nested ones indented:
      CompileUserCall *)

      (** Compiles a call of a declared procedure: its arguments, by value or
          VAR, and the call.
          Also: CALLNONSPECIAL (UCSD20) *)
      PROCEDURE CompileUserCall;
        LABEL 1;
        VAR
          (** The arguments. *)
          ArgumentCount: INTEGER;
          (** The next parameter. *)
          Parameter: LinkedSymbol;
          (** Its type. *)
          ExpectedType: Structure;
          (** The argument is a character constant. *)
          LiteralArgument: BOOLEAN;
          (** It is not a variable of the parameter's size. *)
          SizeDiffers: BOOLEAN;
          (** The parameter's highest index. *)
          Highest: INTEGER;
          (** Its lowest. *)
          Lowest: INTEGER;
          (** The argument's constant value. *)
          LiteralValue: WordValue;
      BEGIN
        WITH Called^ DO
        BEGIN
          Parameter := NextEntry;
          IF StaticLevel = 1 THEN
            IF RoutineKind <> 0 THEN
              Error(399)
        END;
        IF Token = LeftParenthesisToken THEN
        BEGIN
          REPEAT
            IF Parameter = NIL THEN
              Error(126);
            NextToken;
            CompileExpression(Followers + [CommaToken, RightParenthesisToken]);
            LiteralValue.Entry := Operand.Literal.Entry;
            IF (Operand.DataType <> NIL) AND (Parameter <> NIL) THEN
            BEGIN
              ExpectedType := Parameter^.DataType;
              IF (Parameter^.Class = FormalClass) OR (ExpectedType <> NIL) THEN
              BEGIN
                IF Parameter^.Class = ActualClass THEN
                  IF Operand.DataType^.Form <= SetForm THEN
                  BEGIN
                    LiteralArgument := (Operand.DataType = CharStructure) AND
                        (Operand.Kind = KnownValueOperand);
                    SizeDiffers := (ExpectedType^.Size <>
                        Operand.DataType^.Size) OR (Operand.Kind <>
                        MemoryOperand);
                    PushOperand;
                    IF ExpectedType^.Form = SetForm THEN
                      IF SizeDiffers THEN
                        EmitOneOperand(AdjustSetCode, ExpectedType^.Size)
                      ELSE
                      BEGIN
                        CodeCounter := CodeCounter - 1;
                        IF ExpectedType^.Size > 127 THEN
                          CodeCounter := CodeCounter - 2
                      END
                    ELSE
                      IF ExpectedType^.Form = LongIntegerForm THEN
                      BEGIN
                        IF Operand.DataType = IntegerStructure THEN
                        BEGIN
                          EmitConstant(18);
                          EmitExternalCall(30, 4);
                          Operand.DataType := LongStructure
                        END;
                        IF SizeDiffers THEN
                        BEGIN
                          EmitConstant(ExpectedType^.Size);
                          EmitConstant(0);
                          EmitExternalCall(30, 4)
                        END
                        ELSE
                          CodeCounter := CodeCounter - 1
                      END
                      ELSE
                        IF (ExpectedType^.Form = SubrangeForm) AND CheckRanges
                            THEN
                        BEGIN
                          EmitConstant(ORD(ExpectedType^.ArrayIsPacked));
                          EmitConstant(ExpectedType^.MaximumValue);
                          EmitOpcode(RangeCheckCode)
                        END
                        ELSE
                          IF (Operand.DataType = IntegerStructure) AND
                              CompatibleTypes(ExpectedType, RealStructure) THEN
                          BEGIN
                            EmitOpcode(FloatTopCode);
                            Operand.DataType := RealStructure
                          END
                          ELSE
                            IF LiteralArgument AND IsString(ExpectedType) THEN
                              Operand.DataType := StringStructure
                  END
                  ELSE
                  BEGIN
                    LiteralArgument := IsString(Operand.DataType) AND
                        (Operand.Kind = KnownValueOperand);
                    PushAddress;
                    IF LiteralArgument AND IsPackedChars(ExpectedType) THEN
                      IF NOT ExpectedType^.ArrayIsString THEN
                      BEGIN
                        PackStringConstant(LastStringOffset);
                        IF ExpectedType^.BaseType <> NIL THEN
                        BEGIN
                          GetBounds(ExpectedType^.BaseType, Lowest, Highest);
                          IF Highest - Lowest + 1 <> Operand.DataType^.Capacity
                              THEN
                            Error(142)
                        END;
                        Operand.DataType := ExpectedType
                      END
                  END
                ELSE
                  IF Operand.Kind = MemoryOperand THEN
                  BEGIN
                    IF Operand.Literal.Ordinal = ByteAccess THEN
                      IF ExpectedType = ByteStreamStructure THEN
                      BEGIN
                        EmitOpcode(AddIntegersCode);
                        Operand.Literal.Ordinal := IndirectAccess;
                        Operand.StaticLevel := 0
                      END
                      ELSE
                        Error(103);
                    PushAddress;
                    IF ExpectedType <> NIL THEN
                      WITH ExpectedType^ DO
                        CASE Form OF
                          LongIntegerForm, SetForm:
                            IF Operand.DataType^.Size <> ExpectedType^.Size THEN
                              Error(142);
                          ArrayForm:
                            IF ArrayIsPacked THEN
                              IF ArrayIsString THEN
                                IF (Capacity > Operand.DataType^.Capacity) AND
                                    CheckVarStrings THEN
                                  Error(175)
                        END
                  END
                  ELSE
                    Error(154);
                IF NOT CompatibleTypes(ExpectedType, Operand.DataType) THEN
                  IF (ExpectedType <> ByteStreamStructure) AND (ExpectedType <>
                      WordStreamStructure) THEN
                    Error(142)
                  ELSE
                    WITH Operand.DataType^ DO
                      IF ExpectedType = ByteStreamStructure THEN
                        IF (Form = ArrayForm) AND ArrayIsPacked AND
                            (ElementsPerWord = 2) AND (ElementBits = 8) THEN
                        BEGIN
                        END
                        ELSE
                          IF (Operand.DataType = CharStructure) AND
                              (LiteralValue.Ordinal = 4) THEN
                          BEGIN
                          END
                          ELSE
                            IF (Form = SubrangeForm) AND (MaximumValue = 255)
                                THEN
                            BEGIN
                            END
                            ELSE
                              Error(142)
                      ELSE
                        IF (Form = ArrayForm) AND NOT ArrayIsPacked THEN
                        BEGIN
                        END
                        ELSE
                          IF Size = 1 THEN
                          BEGIN
                          END
                          ELSE
                            Error(142)
              END
            END;
            IF Parameter <> NIL THEN
              Parameter := Parameter^.Next
          UNTIL Token <> CommaToken;
          IF Token = RightParenthesisToken THEN
            NextToken
          ELSE
            Error(4)
        END;
        IF Parameter <> NIL THEN
          Error(126);
        WITH Called^ DO
          IF StaticLevel = 1 THEN
          BEGIN
            IF Class = ValueFunctionClass THEN
            BEGIN
              EmitConstant(0);
              EmitConstant(0)
            END;
            IF RoutineNumber = 0 THEN
              NumberExternalRoutine(Called);
            IF InUnit THEN
              IF IsImported THEN
              BEGIN
                EmitTwoOperands(CallExternalCode, CodeSegment, RoutineNumber);
                AddLinkerReference(5, CodeSegment, CodeCounter - 2)
              END
              ELSE
                GOTO 1
            ELSE
            BEGIN
              1:
              IF CodeSegment <> CurrentSegment THEN
                EmitTwoOperands(CallExternalCode, CodeSegment, RoutineNumber)
              ELSE
                IF Offset = 0 THEN
                  EmitOneOperand(CallBaseCode, RoutineNumber)
                ELSE
                  IF Offset = Level THEN
                    EmitOneOperand(CallLocalCode, RoutineNumber)
                  ELSE
                    IF Offset = 1 THEN
                      EmitOneOperand(CallGlobalCode, RoutineNumber)
                    ELSE
                      EmitOneOperand(CallOuterCode, RoutineNumber)
            END
          END
          ELSE
            IF (Offset <> 21) AND (Offset <> 22) THEN
              EmitOneOperand(CallStandardCode, Offset);
        Operand.DataType := Called^.DataType
      END;

    BEGIN
      IF Called^.StaticLevel = 2 THEN
      BEGIN
        HasParentheses := TRUE;
        Key := Called^.Offset;
        IF Token = LeftParenthesisToken THEN
          NextToken
        ELSE
          IF Key IN [ReadlnKey, WritelnKey..LineEndKey] THEN
            HasParentheses := FALSE
          ELSE
            Error(9);
        IF Key IN [PredecessorKey..AbsoluteKey, ReadUnitKey, WriteUnitKey,
            TreeLookupKey, ScanKey, TruncateKey, StrKey] THEN
        BEGIN
          CompileExpression(Followers + [CommaToken, RightParenthesisToken]);
          PushOperand
        END;
        IF Key IN [ReadKey, ReadlnKey] THEN
          CompileInput
        ELSE
          IF Key IN [WriteKey, WritelnKey] THEN
            CompileOutput
          ELSE
            ROUTINE(Followers, HasParentheses, Key);
        IF HasParentheses THEN
          IF Token = RightParenthesisToken THEN
            NextToken
          ELSE
            Error(4)
      END
      ELSE
        CompileUserCall
    END;
