# The 6502 family's instruction-set description: what the assembler, disassembler and tools know about
# these CPUs, and the opcode maps the cores are generated from.
#
# [isa] and [forms] describe the family (read by Retrocore.Isa.IsaParser). Each CPU section lists
# 'opcode Instruction Form', optionally followed by 'undocumented'. For the cores, Instruction and Form are
# members of the Retrocore.Cpu.M6502 Instruction and AddressMode enums. Base sections define every opcode from
# 00 to FF exactly once; derived sections inherit an earlier complete section, then override some entries.

[isa]
endian little
address-bits 16
registers A:8 X:8 Y:8 S:8 P:8 PC:16
flags P N=7 V=6 D=3 I=2 Z=1 C=0
# Documentation names flags by these words on every CPU, never by letter: Out: carry = found.
flag N negative "Set when a result's bit 7 is 1, a negative number in two's complement. BIT copies the tested byte's bit 7 into it."
flag V overflow "Set when a signed add or subtract overflows past -128 or 127. BIT copies the tested byte's bit 6 into it; CLV clears it."
flag D decimal "Set for binary-coded decimal arithmetic in ADC and SBC, by SED; CLD clears it."
flag I interrupt "Set to mask IRQ interrupts, by SEI and as an interrupt starts; NMI and BRK still run. CLI clears it."
flag Z zero "Set when a result is zero, or when a compare finds the values equal."
flag C carry "Set when an add carries out of bit 7, or a subtract or compare does not borrow (the register is at least the value). Shifts and rotates move bits through it."
bit-suffix BBR BBS RMB SMB          # these mnemonics end in the bit they test or change, opcode bits 4-6
prefer EA                           # NOP's official encoding wins over the undocumented ones before it
subset 65SC02 of Wdc65C02 exclude WAI STP BBR BBS RMB SMB message "Instruction '{0}' is not in the 65SC02 subset."
indirect-page-wrap Mos6502          # NMOS JMP ($xxFF) reads its high byte from $xx00
vectors FFFA FFFC FFFE              # NMI, RESET and IRQ/BRK, which analysis follows in this order

# Control flow, for disassembly and analysis. Mnemonic:Form entries override the mnemonic's own class.
flow branch BBR BBS BCC BCS BEQ BMI BNE BPL BVC BVS
flow jump BRA JMP
flow indirect-jump JMP:AbsoluteIndirect JMP:AbsoluteIndexedIndirect
flow call JSR
flow return RTS RTI
flow break BRK
flow wait WAI
flow stop STP JAM

# How an instruction accesses a memory operand; the rest read it.
access write STA STX STY STZ SAX AHX SHX SHY TAS
access read-write ASL LSR ROL ROR INC DEC TRB TSB RMB SMB SLO RLA SRE RRA DCP ISC

[forms]
# Operand forms, in the order operands are matched. A pattern is literal text with {} for an expression.
# Literal forms match only when the mnemonic has them. Forms sharing a pattern are alternatives; a narrow-of
# form is the narrow (zero-page) alternative its value may shrink to. A narrow form takes only a narrow
# address. An exclusive form takes any operand of a mnemonic that has it. A memory form's first field is an
# address the instruction accesses; a pointer form's is the address of a pointer it reads, and an indexed
# pointer form adds an index register to that address first.
Implied                         ""
Stack                           ""
Accumulator                     "A" ""
Immediate                       "#{}"       fields=byte
ZeroPageRelative                "{}"        fields=byte,rel8   exclusive memory message="Bit branches need zero-page address and target."
ProgramCounterRelative          "{}"        fields=rel8        exclusive
ZeroPageIndirectIndexedY        "({}),Y"    fields=byte        narrow pointer
ZeroPageIndexedIndirect         "({},X)"    fields=byte        narrow pointer indexed
AbsoluteIndexedIndirect         "({},X)"    fields=word        pointer indexed
ZeroPageIndirect                "({})"      fields=byte        narrow pointer
AbsoluteIndirect                "({})"      fields=word        pointer
AbsoluteIndexedX                "{},X"      fields=word        memory
ZeroPageIndexedX                "{},X"      fields=byte        narrow-of=AbsoluteIndexedX memory
AbsoluteIndexedY                "{},Y"      fields=word        memory
ZeroPageIndexedY                "{},Y"      fields=byte        narrow-of=AbsoluteIndexedY memory
Absolute                        "{}"        fields=word        memory
ZeroPage                        "{}"        fields=byte        narrow-of=Absolute memory
