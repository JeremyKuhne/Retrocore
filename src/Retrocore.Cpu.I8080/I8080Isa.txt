# The Intel 8080 and 8085 instruction-set description: what the assembler, disassembler and tools know about
# them. The two share one opcode map; the 8085 adds RIM and SIM where the 8080 has undocumented NOPs, and its
# undocumented instructions (DSUB, ARHL, RDEL, LDHI, LDSI, RSTV, SHLX, LHLX, JNK and JK) where the 8080 has
# undocumented aliases of JMP, RET and CALL. Their timings differ, so each set lists every opcode.
#
# [isa] and [forms] describe the CPUs (read by Retrocore.Isa.IsaParser). Each opcode line lists the opcode, the
# instruction and its operand form, 't=' with the nominal states (taken/not taken for a conditional
# instruction) and optionally 'undocumented'. A form the [forms] section does not declare is a literal form
# whose pattern is its name, such as A,M or PSW.
#
# The native syntax is Intel's: M is the byte at HL, a register pair is named by its first register (B, D, H)
# or SP and PSW, and RST takes the restart number 0-7.

[isa]
endian little
address-bits 16
registers A:8 F:8 B:8 C:8 D:8 E:8 H:8 L:8 BC:16 DE:16 HL:16 PSW:16 SP:16 PC:16
flags F S=7 Z=6 K=5 AC=4 P=2 V=1 CY=0
# Documentation names flags by these words on every CPU, never by letter: Out: carry = found. On the 8080 bit
# 5 and bit 3 read 0 and bit 1 reads 1; the 8085's undocumented K (bit 5) and V (bit 1) have no words.
flag S negative "Set when a result's bit 7 is 1, a negative number in two's complement. Intel calls it the sign flag."
flag Z zero "Set when a result is zero, or when a compare finds the values equal."
flag AC half-carry "Auxiliary carry: set when a result carries out of bit 3. DAA uses it to correct decimal results. ANA sets it from bit 3 of the operands ORed on the 8080, and always on the 8085."
flag P parity "Set when a result has an even number of 1 bits."
flag CY carry "Set when an add carries out of the top bit, or a subtract or compare borrows (A is less than the value). Rotates move bits through it."
literal-forms
# The restarts, and on the 8085 alone the entries of TRAP and RST 5.5, 6.5 and 7.5, which on the 8080 fall inside
# the restarts' own code.
entry-points 0000 0008 0010 0018 0020 0028 0030 0038
entry-points in I8085 0024 002C 0034 003C

# Control flow, for disassembly and analysis. A conditional return continues when it is not taken, so it keeps
# the default class.
flow branch JNZ JZ JNC JC JPO JPE JP JM JNK JK
flow jump JMP
flow indirect-jump PCHL
# RSTV (8085) calls $40 when V is set; the opcode implies its target.
flow call CALL CNZ CZ CNC CC CPO CPE CP CM RST RSTV
target RSTV 0040
flow return RET
flow wait HLT

# How an instruction accesses a memory operand; the rest read it.
access write STA SHLD

[forms]
# Operand forms with expressions, in the order operands are matched; literal forms such as A,B and M match
# first, when the mnemonic has them. Forms sharing a pattern are alternatives the mnemonic picks between. An
# exclusive form takes any operand of a mnemonic that has it. A memory form's first field is an address the
# instruction accesses. Fields: n byte, nn word, and rst a restart number 0-7 held in the opcode.
Implied                         ""
rst                             "{}"            fields=rstn exclusive
B,n                             "B,{}"          fields=byte
C,n                             "C,{}"          fields=byte
D,n                             "D,{}"          fields=byte

D,nn                            "D,{}"          fields=word
