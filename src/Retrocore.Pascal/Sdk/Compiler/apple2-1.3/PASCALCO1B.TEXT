  (** Assembled: scans the identifier at SourceBuffer^[SourceCursor] into
      TokenName in capitals, and looks it up in the reserved words, setting
      Token and Operator (II.0: IDSEARCH, here in 6502 code). *)
  PROCEDURE ScanIdentifier(VAR Cursor: INTEGER; Buffer: SourceReference);
    EXTERNAL;

  (** Assembled: searches a scope's tree for a name; 0 found, 1 to go right,
      -1 left, with the last node looked at (II.0: TREESEARCH, here in 6502
      code; PS, TRS). *)
  FUNCTION SearchTree(Root: Identifier; VAR Found: Identifier; VAR SoughtName:
      Spelling): INTEGER;
    EXTERNAL;
