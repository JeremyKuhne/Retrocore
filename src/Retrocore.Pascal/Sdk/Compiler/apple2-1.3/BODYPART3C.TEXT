    (* Compiles a variable *)
    PROCEDURE CompileVariable;
      VAR
        (** UCSD Pascal II.0 calls it LCP (UCSD20). *)
        Found: SymbolLink;
    BEGIN
      IF Token = NameToken THEN
      BEGIN
        FindIdentifier(ClassesOfVariable + [FieldClass], Found);
        NextToken
      END
      ELSE
      BEGIN
        Error(2);
        Found.Entry := DummyVariable.Entry
      END;
      ApplySelectors(Followers, Found.Entry)
    END;

    (* Gives an external procedure the next procedure number *)
    PROCEDURE NumberExternalRoutine;
    BEGIN
      WITH Unnumbered^ DO
      BEGIN
        RoutineNumber := NextRoutine;
        IF NextRoutine > 254 THEN
          Error(251)
        ELSE
          NextRoutine := NextRoutine + 1;
        RoutineDictionary[RoutineNumber] := 0;
        NeedsLinker := TRUE;
        IsExternal := TRUE
      END
    END;

    (* Loads a variable's address: LDA for one of its own, LOD for a VAR
       parameter's *)
    PROCEDURE PushVariableAddress;
    BEGIN
      WITH Variable^ DO
        IF Class = ActualClass THEN
          EmitTwoOperands(OuterAddressCode, StaticLevel, Offset)
        ELSE
          EmitTwoOperands(LoadOuterCode, StaticLevel, Offset)
    END;

    (* Compiles a call: of a special procedure, READ or WRITE, or of a
       declared one *)
    PROCEDURE CompileCall;
      VAR
        (** A special procedure's number.
            Also: LKEY (UCSD20) *)
        Key: INTEGER;
        (** The call has a parenthesized list. *)
        HasParentheses: BOOLEAN;

      (** Compiles READ and READLN. *)
      PROCEDURE CompileInput;
        VAR
          (** The variable. *)
          Found: SymbolLink;
          (** The file read. *)
          InputFile: SymbolLink;
      BEGIN
        InputFile.Entry := InputIdentifier.Entry;
        IF (Token = NameToken) AND HasParentheses THEN
        BEGIN
          FindIdentifier(ClassesOfVariable + [FieldClass], Found);
          IF Found.Entry^.DataType <> NIL THEN
            IF Found.Entry^.DataType^.Form = FileForm THEN
              IF Found.Entry^.DataType^.BaseType = CharStructure THEN
              BEGIN
                NextToken;
                InputFile.Entry := Found.Entry;
                IF NOT (Token IN [CommaToken, RightParenthesisToken]) THEN
                  Error(20);
                IF Token = CommaToken THEN
                  NextToken
              END
        END
        ELSE
          IF HasParentheses THEN
            Error(2);
        IF HasParentheses AND (Token <> RightParenthesisToken) THEN
          REPEAT
            PushVariableAddress(InputFile.Entry);
            CompileVariable(Followers + [CommaToken, RightParenthesisToken]);
            IF Operand.Literal.Ordinal = ByteAccess THEN
              Error(103);
            PushAddress;
            IF Operand.DataType <> NIL THEN
              IF CompatibleTypes(IntegerStructure, Operand.DataType) THEN
                EmitTwoOperands(CallExternalCode, 0, 12)
              ELSE
                IF CompatibleTypes(RealStructure, Operand.DataType) THEN
                  EmitExternalCall(31, 3)
                ELSE
                  IF CompatibleTypes(LongStructure, Operand.DataType) THEN
                  BEGIN
                    EmitConstant(Operand.DataType^.Size);
                    EmitExternalCall(30, 2)
                  END
                  ELSE
                    IF CompatibleTypes(CharStructure, Operand.DataType) THEN
                      EmitTwoOperands(CallExternalCode, 0, 16)
                    ELSE
                      IF IsString(Operand.DataType) THEN
                      BEGIN
                        EmitConstant(Operand.DataType^.Capacity);
                        EmitTwoOperands(CallExternalCode, 0, 18)
                      END
                      ELSE
                        Error(125);
            IF CheckInputOutput THEN
              EmitOneOperand(CallStandardCode, 0);
            LoopDone := Token <> CommaToken;
            IF NOT LoopDone THEN
              NextToken
          UNTIL LoopDone;
        IF Key = ReadlnKey THEN
        BEGIN
          PushVariableAddress(InputFile.Entry);
          EmitTwoOperands(CallExternalCode, 0, 21);
          IF CheckInputOutput THEN
            EmitOneOperand(CallStandardCode, 0)
        END
      END;

      (** Compiles WRITE and WRITELN. *)
      PROCEDURE CompileOutput;
        VAR
          (** The value's type. *)
          ValueType: Structure;
          (** Its field width is the default. *)
          DefaultWidth: BOOLEAN;
          (** The first argument. *)
          Found: SymbolLink;
          (** The file written. *)
          OutputFile: SymbolLink;
          (** A string's length. *)
          ArrayLength: INTEGER;
          (** A packed array's lowest index. *)
          Lowest: INTEGER;
          (** A word CompileOutput never uses. *)
          UnusedWord: INTEGER;
      BEGIN
        OutputFile.Entry := OutputIdentifier.Entry;
        IF (Token = NameToken) AND HasParentheses THEN
        BEGIN
          FindIdentifier(ClassesOfVariable + [ConstantClass, FieldClass,
              ValueFunctionClass], Found);
          IF Found.Entry^.DataType <> NIL THEN
            IF Found.Entry^.DataType^.Form = FileForm THEN
              IF Found.Entry^.DataType^.BaseType = CharStructure THEN
              BEGIN
                NextToken;
                OutputFile.Entry := Found.Entry;
                IF NOT (Token IN [CommaToken, RightParenthesisToken]) THEN
                  Error(20);
                IF Token = CommaToken THEN
                  NextToken
              END
        END;
        IF HasParentheses AND (Token <> RightParenthesisToken) THEN
          REPEAT
            PushVariableAddress(OutputFile.Entry);
            CompileExpression(Followers + [CommaToken, ColonToken,
                RightParenthesisToken]);
            ValueType := Operand.DataType;
            IF ValueType <> NIL THEN
              WITH ValueType^ DO
                IF Form > LongIntegerForm THEN
                  PushAddress
                ELSE
                BEGIN
                  PushOperand;
                  IF Form = LongIntegerForm THEN
                  BEGIN
                    EmitConstant(DecimalSize(36));
                    EmitConstant(0);
                    EmitExternalCall(30, 4)
                  END
                END;
            IF Token = ColonToken THEN
            BEGIN
              NextToken;
              CompileExpression(Followers + [CommaToken, ColonToken,
                  RightParenthesisToken]);
              IF Operand.DataType <> NIL THEN
                IF Operand.DataType <> IntegerStructure THEN
                  Error(20);
              PushOperand;
              DefaultWidth := FALSE
            END
            ELSE
              DefaultWidth := TRUE;
            IF ValueType = IntegerStructure THEN
            BEGIN
              IF DefaultWidth THEN
                EmitConstant(0);
              EmitTwoOperands(CallExternalCode, 0, 13)
            END
            ELSE
              IF ValueType = RealStructure THEN
              BEGIN
                IF DefaultWidth THEN
                  EmitConstant(0);
                IF Token = ColonToken THEN
                BEGIN
                  NextToken;
                  CompileExpression(Followers + [CommaToken,
                      RightParenthesisToken]);
                  PushOperand;
                  IF Operand.DataType <> NIL THEN
                    IF Operand.DataType <> IntegerStructure THEN
                      Error(125)
                END
                ELSE
                  EmitConstant(0);
                EmitExternalCall(31, 4)
              END
              ELSE
                IF CompatibleTypes(ValueType, LongStructure) THEN
                BEGIN
                  IF DefaultWidth THEN
                    EmitConstant(0);
                  EmitExternalCall(30, 3)
                END
                ELSE
                  IF ValueType = CharStructure THEN
                  BEGIN
                    IF DefaultWidth THEN
                      EmitConstant(0);
                    EmitTwoOperands(CallExternalCode, 0, 17)
                  END
                  ELSE
                    IF IsString(ValueType) THEN
                    BEGIN
                      IF DefaultWidth THEN
                        EmitConstant(0);
                      EmitTwoOperands(CallExternalCode, 0, 19)
                    END
                    ELSE
                      IF IsPackedChars(ValueType) AND (ValueType <>
                          ByteStreamStructure) THEN
                      BEGIN
                        ArrayLength := 0;
                        IF ValueType^.BaseType <> NIL THEN
                        BEGIN
                          GetBounds(ValueType^.BaseType, Lowest, ArrayLength);
                          ArrayLength := ArrayLength - Lowest + 1
                        END;
                        IF DefaultWidth THEN
                          EmitConstant(ArrayLength);
                        EmitConstant(ArrayLength);
                        EmitTwoOperands(CallExternalCode, 0, 20)
                      END
                      ELSE
                        Error(125);
            IF CheckInputOutput THEN
              EmitOneOperand(CallStandardCode, 0);
            LoopDone := Token <> CommaToken;
            IF NOT LoopDone THEN
              NextToken
          UNTIL LoopDone;
        IF Key = WritelnKey THEN
        BEGIN
          PushVariableAddress(OutputFile.Entry);
          EmitTwoOperands(CallExternalCode, 0, 22);
          IF CheckInputOutput THEN
            EmitOneOperand(CallStandardCode, 0)
        END
      END;
